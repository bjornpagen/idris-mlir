```idris
data Mark = Unseen | Open Nat | Done

one : Mark -> Nat
one Unseen = 1
one _ = 0

unseen : Vect n Mark -> Nat
unseen [] = 0
unseen (m :: ms) = one m + unseen ms

data Look : Vect n Mark -> Type where
  Fresh : {0 ms : Vect n Mark} -> (ms' : Vect n Mark) -> (0 _ : LT (unseen ms') (unseen ms)) -> Look ms
  OnStack : {0 ms : Vect n Mark} -> Nat -> Look ms
  Finished : {0 ms : Vect n Mark} -> Look ms

0 consLT : (m : Mark) -> LT a b -> LT (one m + a) (one m + b)
consLT m lt = rewrite plusSuccRightSucc (one m) a in plusLteMonotoneLeft (one m) (S a) b lt

look : (v : Fin n) -> (i : Nat) -> (ms : Vect n Mark) -> Look ms
look FZ i (Unseen :: ms) = Fresh (Open i :: ms) reflexive
look FZ i (Open j :: ms) = OnStack j
look FZ i (Done :: ms) = Finished
look (FS v) i (m :: ms) = case look v i ms of
  Fresh ms' lt => Fresh (m :: ms') (consLT m lt)
  OnStack j => OnStack j
  Finished => Finished

0 doneNoMore : (w : Fin n) -> (ms : Vect n Mark) -> LTE (unseen (replaceAt w Done ms)) (unseen ms)
doneNoMore FZ (m :: ms) = rewrite plusCommutative (one m) (unseen ms) in lteAddRight (unseen ms)
doneNoMore (FS w) (m :: ms) = plusLteMonotoneLeft (one m) _ _ (doneNoMore w ms)

record Search (n : Nat) where
  constructor MkSearch
  next : Nat
  marks : Vect n Mark
  stack : List (Fin n)
  found : List (List (Fin n))

Sized (Search n) where
  size s = unseen s.marks

0 NoMore : Search n -> Search n -> Type
NoMore s' s = LTE (size s') (size s)

finish : List (Fin n) -> (ms : Vect n Mark) -> Subset (Vect n Mark) (\ms' => LTE (unseen ms') (unseen ms))
finish [] ms = Element ms reflexive
finish (w :: ws) ms = let Element ms1 p1 = finish ws ms in Element (replaceAt w Done ms1) (transitive (doneNoMore w ms1) p1)

close : (v : Fin n) -> (s : Search n) -> Subset (Search n) (\s' => NoMore s' s)
close v s =
  let (above, rest) = break (== v) s.stack
      members = v :: above
      Element ms p = finish members s.marks in
  Element (MkSearch s.next ms (drop 1 rest) (members :: s.found)) p

entered : (s : Search n) -> (v : Fin n) -> (ms : Vect n Mark) -> Search n
entered s v ms = MkSearch (S s.next) ms (v :: s.stack) s.found

mutual
  visit : Vect n (List (Fin n)) -> (v : Fin n) -> (i : Nat) -> (s : Search n) -> (0 _ : SizeAccessible s) ->
          (Nat, Subset (Search n) (\s' => NoMore s' s))

    (low, Element s1 p1) =>
      if low == i
        then case close v s1 of Element s2 p2 => (low, Element s2 (transitive p2 p1))
        else (low, Element s1 p1)

         (s0 : Search n) -> (0 _ : SizeAccessible s0) -> (s : Search n) -> (0 _ : NoMore s s0) ->
         (Nat, Subset (Search n) (\s' => NoMore s' s))

  arcs adj low (w :: ws) s0 acc@(Access rec) s p = case look w s.next s.marks of
```

```idris
settled : (g : Graph) -> (Sorted g (everyArc g), State g)
settled g =
  let arcs = sortArcs g (everyArc g)
      cs = counted 0 (components arcs.links)
      comp = foldl (\v, (k, ms) => foldl (\v', p => replaceAt p k v') v ms) (the (Vect g.size Nat) (replicate g.size 0)) cs in
  (arcs, foldl (settle g arcs comp) (MkState arcs.floor (replicate g.size False) []) cs)
  where
    counted : Nat -> List a -> List (Nat, a)
    counted k [] = []
    counted k (x :: xs) = (k, x) :: counted (S k) xs

export
dagPotential : (g : Graph) -> Vect g.size Nat
dagPotential g = (snd (settled g)).pot

export
check : (g : Graph) -> Verdict g
check g =
  let (arcs, st) = settled g in
  case (reverse st.cycles, arcs.stucks) of
    (c :: rest, None _) => Grows (c ::: rest)
    (c :: rest, Some ss) => Grows (c ::: rest ++ mapMaybe (cause st.reached) (forget ss))
    ([], Some ss) => Grows (map snd ss)

      Holds f => Bounded st.pot f
      Fails _ => case heaviest g (everyArc g) fin st.pot of
        Left (c ** as ** pf) => Grows (Cycle c as pf ::: [])
        Right (pot ** fits) => Bounded pot (everyFits fits)
```
