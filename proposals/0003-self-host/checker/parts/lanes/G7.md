  > The positions of a type constructor's arguments the growth check watches: its type parameters (`typeParams`, the keys), and every other argument whose kind is a universe and that its values hold. A value holds such an argument when a constructor that has its own binder there holds a runtime argument whose type reaches that binder, through a function type's argument or result, a suspension, or a represented position of a type constructor (this one's, or another's type parameters). It is the least such set: `L : a -> T a` makes `T`'s `a` a position, and `PCons : P (a, a) -> P a`, whose index no value holds, does not. An index is checked by the growth phase, never a key.

  - Another type constructor's represented positions are read as its `typeParams`. Its own held indices would need a fixpoint across types. None is reachable in the tests or the self compile (measured). One that was lost would leave an index unwatched; translation refuses its flow into a key as `dependent field` (`walk`, 705-706), and never keys it.

  ```idris
  export
  represented : {auto c : Ref Ctxt Defs} -> GlobalDef -> Core (List Nat)
  represented def = case definition def of
    TCon arity _ _ _ _ cons _ => do
      ps <- dataParams def
      keys <- typeParams def ps
      defs <- get Ctxt
      ty <- normaliseClosed (type def)
      let values = kinds ty
      -- Every argument position: `[0 .. minus arity 1]` would be `[0]` at
      -- arity 0.
      let candidates = filter (\i => not (elem i ps) && not (fromMaybe False (getAt i values)))
                              (iterateN arity S 0)
      tys <- traverse (\n => map (map type) (lookupCtxtExact n (gamma defs))) (fromMaybe [] cons)
      conTys <- traverse (\t => normaliseClosed t >>= toFullNames) (catMaybes tys)
      held <- rounds (length candidates) keys candidates conTys []
      pure (sort (keys ++ held))
    _ => pure []
    where
      -- The type parameters of another type constructor.
      keysOf : Name -> Core (List Nat)
      keysOf d = do
        defs <- get Ctxt
        Just ddef <- lookupCtxtExact d (gamma defs)
          | Nothing => pure []
        typeParams ddef !(dataParams ddef)

      mutual
        -- Does a type reach the stand-in for binder `k`, at the positions
        -- `own` of this type constructor?
        reaches : List Nat -> Nat -> TT vs -> Core Bool
        reaches own k (Ref _ _ (MN "idris-mlir-binder" i)) = pure (cast i == k)
        reaches own k (Bind _ _ (Pi _ rig _ a) sc) = do
          inA <- if isErased rig then pure False else reaches own k a
          if inA then pure True else reaches own k sc
        reaches own k (TDelayed _ _ t) = reaches own k t
        reaches own k tm = case getFnArgs tm of
          (Ref _ (TyCon _) d, args) => do
            ks <- if d == fullname def then pure own else keysOf d
            spine ks own k (length args) tm
          _ => pure False

        -- The arguments at positions `ks` of an application of `j`
        -- arguments, the last first, so that each is a subterm.
        spine : List Nat -> List Nat -> Nat -> Nat -> TT vs -> Core Bool
        spine ks own k (S j) (App _ f a) = do
          here <- if elem j ks then reaches own k a else pure False
          if here then pure True else spine ks own k j f
        spine _ _ _ _ _ = pure False

      -- Does a constructor whose return type has its own binder at `p` hold
      -- a runtime argument whose type reaches it?
      holds : List Nat -> Nat -> ClosedTerm -> Core Bool
      holds own p conTy =
        let (bs, ret) = conBinders none 0 conTy in
        case getAt p (snd (getFnArgs ret)) of
          Just (Ref _ _ (MN "idris-mlir-binder" k)) =>
            any id <$> traverse (\(rig, a) => if isErased rig then pure False else reaches own (cast k) a) bs
          _ => pure False

      -- The least set, from none: each round that changes it adds a
      -- candidate, so `length candidates` rounds reach it.
      rounds : Nat -> List Nat -> List Nat -> List ClosedTerm -> List Nat -> Core (List Nat)
      rounds Z _ _ _ held = pure held
      rounds (S n) keys candidates conTys held = do
        next <- filterM (\p => any id <$> traverse (holds (keys ++ held) p) conTys) candidates
        if length next == length held then pure held else rounds n keys candidates conTys next
  ```

  - `rounds`'s count is no fuel: it is the number of candidates, and each round that changes the set adds one, so the count is reached only once the set is whole.

  ```idris
  ||| The registrations in progress: the bounds the growth check gave
  ||| (`TS.sizes`, which nothing changes while the translation runs), and the
  ||| links of the data instances being registered, the innermost first.
  record Registering where
    constructor MkRegistering
    sizes : Bounds
    links : List (Link Form)

  ||| One list of links is another with one more in front: a link new to it,
  ||| and in the universe of links within the bounds.
  data Pushed : List (Link Form) -> List (Link Form) -> List (Link Form) -> Type where
    Push : (0 _ : Not (Elem l ls)) -> (0 _ : Elem l u) -> Pushed u (l :: ls) ls

  ||| Why registering ends: each data instance registered pushes its link,
  ||| new to the chain and within the bounds, and the universe of links
  ||| within the bounds is finite (`Requests.Universe`), so each push leaves
  ||| less room in it (`Finite.room`).
  0 Descending : Registering -> Type
  Descending reg = Accessible (Pushed (Universe forms addresses reg.sizes)) reg.links

  -- A chain of distinct links of the universe descends: each push leaves
  -- room for fewer.
  0 pushes : (u, ls : List (Link Form)) -> NoDup ls -> All (\l => Elem l u) ls ->
             Accessible LT (minus (length u) (length ls)) -> Accessible (Pushed u) ls
  pushes u ls distinct inside (Access less) =
    Access (\ls', pushed => case pushed of
      Push {l} fresh member =>
        pushes u (l :: ls) (fresh :: distinct) (member :: inside)
               (less _ (room (fresh :: distinct) (member :: inside))))

  -- The empty chain descends.
  0 started : (b : Bounds) -> Descending (MkRegistering b [])
  started b = pushes (Universe forms addresses b) [] [] [] (wellFounded _)
  ```

  ```idris
  ||| at its type constructor's node (`MkNode t []`), checked where the
  ||| instance is registered, on the argument as the key holds it. A
  ||| position given no bound, or an argument above its bound, is a flow the
  ```
