\begin{exmp}[Parametricity Failure]
  This example breaks type soundness and parametricity. However, the embedded
  target term is not the translation of any valid well-typed source term. To
  overcome this problem we can seal/annotate opaque and perform a dynamic
  type check before removing the brackets.
  \[
  \begin{array}{r l}
    & \salien{\tabs{\tvar{x}}{\tabs{\tvar{y}}{\tvar{x}}}}{\sall{\svar{A}}{\sall{\svar{B}}{\sarrow{\svar{A}}{\sarrow{\svar{B}}{\svar{B}}}}}}{\cdot}
    ~\snat~\stunit~\sexpi~\sexpii \\[0.4em]
    \sevalrs{Alien}
    & \salien{\tabs{\tvar{x}}{\tabs{\tvar{y}}{\tvar{x}}}}{\sarrow{\svar{A}}{\sarrow{\svar{B}}{\svar{B}}}}
            {\sesubsttwo{\svar{A}}{\snat}{\svar{B}}{\stunit}}~\sexpi~\sexpii \\[0.4em]
    \sevalrs{Alien + App}
    & \salien{\tabs{\tvar{y}}{\tvar{\topaque{\sexpi}}}}{\sarrow{\svar{B}}{\svar{B}}}
            {\sesubsttwo{\svar{A}}{\snat}{\svar{B}}{\stunit}}~\sexpii \\[0.4em]
    \sevalrs{Alien + App}
    & \salien{\tvar{\topaque{\sexpi}}}{\svar{B}}
            {\sesubsttwo{\svar{A}}{\snat}{\svar{B}}{\stunit}}\\[0.4em]
    \sevalr{Alien}
    & \sexpi
  \end{array}
  \]
  
\end{exmp}
