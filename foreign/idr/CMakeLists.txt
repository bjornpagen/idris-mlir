# idris-mlir-cc links the runtime's bitcode into the program (Linker,
# BitReader, Object), internalizes it (IPO) and writes the runtime it
# prepared (BitWriter); idr-eval compiles with ORC's LLJIT (OrcJIT) and the
# same pipeline (Passes). The runtime's bitcode carries inline assembly (CPU
# feature probes), which the object emitter parses with the native target's
# assembly parser.
llvm_map_components_to_libnames(IDR_LLVM_CODEGEN_LIBS nativecodegen ${LLVM_NATIVE_ARCH} passes linker bitreader bitwriter object ipo orcjit)

# The target entry's lists, as X-macros of C strings in a generated header
# (idr/TargetEntry.h): what links a program after its object, the runtime
# and -o (--target with the triple, the executable flags, the program link
# flags, GMP), which idris-mlir-cc --print-link-flags prints, and the
# library calls idr-eval's JIT binds from the process.
function(idr_strings out)
	set(text "")
	foreach(item IN LISTS ARGN)
		string(REPLACE "\\" "\\\\" item "${item}")
		string(REPLACE "\"" "\\\"" item "${item}")
		string(APPEND text " X(\"${item}\")")
	endforeach()
	set(${out} "${text}" PARENT_SCOPE)
endfunction()
idr_strings(IDR_LINK_FLAGS "--target=${CMAKE_CXX_COMPILER_TARGET}" ${IDRIS_MLIR_EXECUTABLE_FLAGS}
	${IDRIS_MLIR_PROGRAM_LINK_FLAGS} ${IDRIS_MLIR_GMP})
idr_strings(IDR_JIT_LIBRARY_CALLS ${IDRIS_MLIR_JIT_LIBRARY_CALLS})
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/include/idr/TargetEntry.h" CONTENT [=[
// Generated by CMake from the target entry (CMakeLists.txt). Do not edit.
#pragma once

#define IDRIS_MLIR_LINK_FLAGS(X)@IDR_LINK_FLAGS@
#define IDRIS_MLIR_JIT_LIBRARY_CALLS(X)@IDR_JIT_LIBRARY_CALLS@
]=] @ONLY)

# The runtime every program links: the archive, checked before idris-mlir-cc
# is built, then prepared once by idris-mlir-cc itself (--prepare-runtime)
# into one object, native code for every executable's link line with the
# same functions as bitcode for inlining, kept as the target entry's
# container says (CMakeLists.txt), which is the runtime the driver links by
# default (--runtime overrides the path, with the archive itself if need be;
# --print-runtime tells the link line where it is).
set(IDRIS_MLIR_RUNTIME_PREPARED "${CMAKE_CURRENT_BINARY_DIR}/idris_rt.o")
target_compile_definitions(idris-mlir-cc PRIVATE
	"IDRIS_MLIR_RUNTIME=\"${IDRIS_MLIR_RUNTIME_PREPARED}\""
	"IDRIS_MLIR_RUNTIME_BITCODE_SECTION=\"${IDRIS_MLIR_RUNTIME_BITCODE_SECTION}\""
)
add_dependencies(idris-mlir-cc idris_rt_check)
set(IDRIS_MLIR_RUNTIME_PREPARED_FILES "${IDRIS_MLIR_RUNTIME_PREPARED}")
if(IDRIS_MLIR_RUNTIME_BITCODE_SECTION STREQUAL "")
	# The bitcode, beside the object.
	list(APPEND IDRIS_MLIR_RUNTIME_PREPARED_FILES "${CMAKE_CURRENT_BINARY_DIR}/idris_rt.bc")
endif()
add_custom_command(
	OUTPUT ${IDRIS_MLIR_RUNTIME_PREPARED_FILES}
	COMMAND idris-mlir-cc --prepare-runtime "--runtime=${IDRIS_MLIR_RUNTIME_ARCHIVE}"
		-o "${IDRIS_MLIR_RUNTIME_PREPARED}"
	DEPENDS idris-mlir-cc idris_rt idris_rt_check
	COMMENT "Optimizing and compiling the runtime once, for every program"
	VERBATIM
)
add_custom_target(idris_rt_prepared ALL DEPENDS "${IDRIS_MLIR_RUNTIME_PREPARED}")

# The MLIR API is not module-aware, so no target imports std: idr.mlir
# wraps its headers instead. Only idr_dialect has module units; the tools
# are plain translation units.
set_target_properties(idr_dialect idris-mlir-opt idris-mlir-cc idris-mlir-reduce idris-mlir-tblgen PROPERTIES
	CXX_MODULE_STD OFF
)
set_target_properties(idr_dialect PROPERTIES CXX_SCAN_FOR_MODULES ON)

# Every plain unit reaches idr/Idr.h, which parses most of MLIR's headers:
# 18 of a unit's 22 seconds. Precompiled once, a unit takes 4. Module units
# parse their headers in their global module fragment, where a
# precompiled header does not belong.
target_precompile_headers(idr_dialect PRIVATE include/idr/Idr.h)
file(GLOB_RECURSE IDR_MODULE_UNITS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/lib/*.cppm)
set_source_files_properties(${IDR_MODULE_UNITS} PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
set_target_properties(idris-mlir-opt idris-mlir-cc idris-mlir-reduce idris-mlir-tblgen PROPERTIES
	CXX_SCAN_FOR_MODULES OFF
)
